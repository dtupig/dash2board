const board = document.getElementById("board");
const countEl = document.getElementById("count");
const wide = window.matchMedia("(min-width: 840px)");

function card(s, aberto){
  const foot = [`dep: ${s.d}`, s.r?`risco: ${s.r}`:null, s.sec?`<span class="sec">commit sec(...)</span>`:null].filter(Boolean).join(" ");
  return `<article${s.key?' class="key"':''}>
    <div class="id-row">
      <span class="id">${s.id}</span>
      <span class="pill p-${s.p[0]}">${s.pl}</span>
      <span class="pts-badge">${s.pts} pt</span>
    </div>
    <h3>${s.t}</h3>
    <p class="story">${s.h}</p>
    <details class="crit"${aberto?" open":""}>
      <summary>${s.ac.length} critérios</summary>
      <ul class="ac-list">${s.ac.map(a=>`<li>${a}</li>`).join("")}</ul>
    </details>
    <div class="foot">${foot}</div>
  </article>`;
}

function render(filtro){
  const aberto = wide.matches;
  board.innerHTML = SPRINTS.map(sp => {
    const itens = S.filter(s => s.s===sp.id && (filtro==="all" || s.p.includes(filtro) || s.p.includes("all")));
    if(!itens.length) return "";
    const pts = itens.reduce((a,s)=>a+s.pts,0);
    return `<section class="sprint">
      <div class="sprint-head">
        <span class="tag">${sp.id}</span>
        <h2>${sp.nome}</h2>
        ${sp.cond?`<span class="cond">${sp.cond}</span>`:""}
        <span class="pts">${pts} pt${filtro==="all"?"":" no filtro"}</span>
      </div>
      <div class="grid">${itens.map(s=>card(s,aberto)).join("")}</div>
    </section>`;
  }).join("");
  const n = board.querySelectorAll("article").length;
  countEl.textContent = `${n}/${S.length} histórias`;
}

let filtroAtual = "all";
document.querySelectorAll("button.f").forEach(b => {
  b.addEventListener("click", () => {
    document.querySelectorAll("button.f").forEach(o => o.setAttribute("aria-pressed", String(o===b)));
    filtroAtual = b.dataset.p;
    render(filtroAtual);
    b.scrollIntoView({block:"nearest", inline:"nearest", behavior:"smooth"});
  });
});
// ao passar de celular para tela larga (girar o aparelho, redimensionar), os critérios abrem
wide.addEventListener("change", () => render(filtroAtual));
render(filtroAtual);
